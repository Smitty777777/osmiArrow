cmake_minimum_required(VERSION 3.20)
project(osmiArrow)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Arrow REQUIRED)
find_package(Parquet REQUIRED)
add_library(osmi_arrow INTERFACE)
target_include_directories(osmi_arrow INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(osmi_arrow INTERFACE Arrow::arrow_shared Parquet::parquet_shared)
add_executable(osmiArrow main.cpp)
target_link_libraries(osmiArrow PRIVATE osmi_arrow)
